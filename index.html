from pathlib import Path p = Path("/mnt/data/klip_blog(1).html") s = p.read_text(encoding="utf-8") # Fix broken HTML quote attributes on feed buttons. s = s.replace('onclick="setFeedMode("latest")"', 'onclick="setFeedMode(\'latest\')"') s = s.replace('onclick="setFeedMode("trending")"', 'onclick="setFeedMode(\'trending\')"') # Use redirect sign-in instead of popup, which avoids the popup immediately closing. s = s.replace( ''' GoogleAuthProvider, signInWithPopup, signOut,''', ''' GoogleAuthProvider, signInWithRedirect, getRedirectResult, signOut,''') old = ''' window.login = async function() { try { await signInWithPopup(auth, provider); status("Signed in!"); } catch(error) { console.error(error); status("Sign-in failed: " + error.message); } };''' new = ''' window.login = async function() { try { await signInWithRedirect(auth, provider); } catch(error) { console.error(error); status("Sign-in failed: " + error.message); } }; // Finish a Google sign-in after Firebase redirects back to Klip. getRedirectResult(auth).catch(error => { console.error("Google sign-in error:", error); status("Sign-in failed: " + error.message); });''' if old not in s: raise RuntimeError("Could not find the login function to replace.") s = s.replace(old, new) out = Path("/mnt/data/index.html") out.write_text(s, encoding="utf-8") print("Created:", out)
